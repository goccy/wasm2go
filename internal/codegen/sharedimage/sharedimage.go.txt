package {{.Pkg}}

// Shared, copy-on-write linear memory.
//
// Most of what an instance of a large wasm module costs is memory that is
// IDENTICAL in every instance and, apart from a handful of pages, never written
// to. A big engine can carry tens of MB of static tables in its data segments,
// and build more heap still while it starts its runtime up. Every instance
// paying for its own copy of all that is pure waste.
//
// So build it ONCE per process and give every instance a private copy-on-write
// map of it: read-only pages stay physically shared, and only a page a guest
// actually writes becomes private.
//
// There are two kinds of image, and they differ in how much of the built
// instance they keep:
//
//   - NewSharedImage keeps the data segments. Instances (NewWithMemory) re-run
//     their own initialization over them, so the image must carry NOTHING above
//     the segments: that tail is BSS, and it holds the start section's
//     "already ran" flag and the C++ static state. Inheriting the flag makes
//     the start section trap on its second run; inheriting statics that point
//     into a heap the image does not carry breaks the first thing that follows
//     one of those pointers. What makes even the segments survive the re-run is
//     that memory.init compares before it writes (see memoryInit) — it finds
//     them already in place and leaves the pages shared.
//
//   - NewSharedSnapshot keeps the whole initialized instance, memory and wasm
//     globals both. Instances (NewFromSnapshot) re-run NOTHING, which is what
//     lets the snapshot keep the BSS and the heap that the first kind has to
//     throw away. It shares much more; it also freezes whatever the build did.
//
// Measured on one large embedding (instances live in a 4 GiB container, created
// until the kernel kills the process): a hundred-odd without any image, several
// times that with the data-segment image, and an order of magnitude more with
// the snapshot, which shares the started runtime's heap on top of the segments.
//
// Where mmap is unavailable (windows, js/wasm — see sharedimage_nommap.go),
// every instance allocates and initializes its own memory, exactly as before.

import (
	"bytes"
	"fmt"
	"math"
	"os"
	"runtime"
	"sync"
)

// SharedImage is a process-wide initialized linear-memory template.
type SharedImage struct {
	f       *os.File // unlinked; holds the image
	size    uint64   // the guest's memory.size for the image
	dataEnd uint64   // where the data segments stop and BSS begins
	globals []uint64 // non-nil for a snapshot: the instance's wasm globals
	err     error    // non-nil if the image could not be built
	// anonymous marks the stand-in that AnonymousMemory returns for memory
	// no file holds: ForkMemory may map that memory onto the image it
	// writes, which it may not do to memory it knows nothing of (an
	// allocation of the Go heap).
	anonymous bool
}

// ImageBuilder brings up one throwaway instance. The caller supplies it because
// only the generated package knows how to construct a Module (its constructor's
// signature depends on the wasm's imports). How far it must initialize that
// instance depends on which of the two images below is being built.
type ImageBuilder func() (m *{{.Module}}, err error)

// NewSharedImage builds the DATA-SEGMENT image once and caches it. Every later
// call returns the same one. A failure is remembered, not retried: callers then
// fall back to a private allocation, which is correct, only larger.
//
// build must run the start section (the ordinary constructor does) and nothing
// more: everything above the data segments is discarded. Instances go through
// NewWithMemory and re-run their own initialization over the shared segments.
func NewSharedImage(build ImageBuilder) *SharedImage {
	sharedOnce.Do(func() { sharedImg = buildSharedImage(build, false) })
	return sharedImg
}

// NewSharedSnapshot builds an image of a FULLY INITIALIZED instance — memory
// and wasm globals, everything, exactly as build() left it — and caches it the
// same way. Instances go through NewFromSnapshot, which re-runs nothing at all.
//
// This shares far more than NewSharedImage does. A runtime's initialization
// (C++ static constructors, an interpreter's own start-up) leaves megabytes of
// heap behind that are identical in every instance, and NewSharedImage cannot
// share ANY of it: it must zero the BSS, because a fresh instance re-runs the
// initialization and would trip over the previous one's leftovers. A snapshot
// re-runs nothing, so it can keep the lot.
//
// The catch is that build() must leave the instance in a state every later
// instance can start from. Anything build() does is done for all of them — an
// instance that is already holding a lock or mid-request is not a snapshot, it
// is a bug. Build it clean and take it at rest.
//
// Unlike NewSharedImage, this does NOT cache: a snapshot bakes in whatever
// build() configured, so whether two callers want the same snapshot is a
// question only the caller can answer. Build it once and hold on to it (a
// caller whose instances come in flavours will want one per flavour); building
// it per instance would be slower than not sharing at all.
func NewSharedSnapshot(build ImageBuilder) *SharedImage {
	return buildSharedImage(build, true)
}

var (
	sharedOnce sync.Once
	sharedImg  *SharedImage

	sharedInPlaceOnce sync.Once
	sharedInPlaceImg  *SharedImage
)

// InPlaceImageBuilder brings up one throwaway instance ON the caller-provided
// memory — a MAP_SHARED mapping of the image file being built, fresh and all
// zero. Install the module's active data segments in it with the generated
// InstallDataSegments, construct the module with the generated NewWithMemory
// (passing InitialMemoryBytes), and initialize it exactly as the
// corresponding ImageBuilder would.
type InPlaceImageBuilder func(mem []byte) (m *{{.Module}}, err error)

// NewSharedImageInPlace is NewSharedImage with the copy elided: the builder
// runs directly on a shared mapping of the image file, so every page it
// touches is written once, straight into the file's page cache — there is no
// intermediate buffer, no copy, and the dirty footprint of building the image
// is exactly the pages the build touches. Cached once per process, like
// NewSharedImage. ceiling is the length of the mapping the builder runs on,
// in bytes (see Memory).
func NewSharedImageInPlace(ceiling uint64, build InPlaceImageBuilder) *SharedImage {
	sharedInPlaceOnce.Do(func() { sharedInPlaceImg = buildSharedImageInPlace(ceiling, build, false) })
	return sharedInPlaceImg
}

// NewSharedSnapshotInPlace is NewSharedSnapshot with the copy elided — see
// NewSharedImageInPlace for the mechanics and NewSharedSnapshot for the
// snapshot contract (build it clean and take it at rest; not cached).
func NewSharedSnapshotInPlace(ceiling uint64, build InPlaceImageBuilder) *SharedImage {
	return buildSharedImageInPlace(ceiling, build, true)
}

func buildSharedImageInPlace(ceiling uint64, build InPlaceImageBuilder, snapshot bool) *SharedImage {
	if !mmapSupported {
		return &SharedImage{err: fmt.Errorf("wasm2go: copy-on-write memory is not supported on this platform")}
	}
	length, err := mappingLength(ceiling)
	if err != nil {
		return &SharedImage{err: err}
	}
	f, err := newTempImageFile()
	if err != nil {
		return &SharedImage{err: fmt.Errorf("wasm2go: shared memory image: %w", err)}
	}
	if err := f.Truncate(int64(length)); err != nil {
		f.Close()
		return &SharedImage{err: fmt.Errorf("wasm2go: sizing the shared memory image: %w", err)}
	}
	mem, err := mmapShared(f, length)
	if err != nil {
		f.Close()
		return &SharedImage{err: err}
	}
	fail := func(err error) *SharedImage {
		unmapMemory(mem)
		f.Close()
		return &SharedImage{err: err}
	}
	m, err := build(mem)
	if err != nil {
		return fail(fmt.Errorf("wasm2go: building the shared memory image: %w", err))
	}
	size := m.{{.MemSize}}.Load()
	dataEnd := m.{{.DataEnd}}
	if dataEnd == 0 || dataEnd > size {
		return fail(fmt.Errorf(
			"wasm2go: shared memory image: no data segments were installed (dataEnd=%d, size=%d)",
			dataEnd, size))
	}
	var globals []uint64
	if snapshot {
		globals = {{.SaveGlobals}}(m)
	}
	if !snapshot {
		// The data-segment kind must carry NOTHING above the segments (see
		// NewSharedImage). The builder's BSS is already in the file, so punch
		// it back out: truncating to dataEnd drops those pages, re-extending
		// restores the sparse zero tail instances expect. The builder ran
		// only the start section and spawned no agents, so nothing touches
		// the mapping above dataEnd after this.
		if err := f.Truncate(int64(dataEnd)); err != nil {
			return fail(fmt.Errorf("wasm2go: trimming the shared memory image: %w", err))
		}
		if err := f.Truncate(int64(length)); err != nil {
			return fail(fmt.Errorf("wasm2go: re-extending the shared memory image: %w", err))
		}
	}
	// The builder's mapping is not GC-managed. A finalizer ties the unmap
	// to the module's reachability: a builder that parked agent goroutines
	// (they reference the module) keeps its mapping until they go.
	runtime.SetFinalizer(m, func(fm *{{.Module}}) { unmapMemory(mem) })
	return &SharedImage{f: f, size: size, dataEnd: dataEnd, globals: globals}
}

func buildSharedImage(build ImageBuilder, snapshot bool) *SharedImage {
	if !mmapSupported {
		return &SharedImage{err: fmt.Errorf("wasm2go: copy-on-write memory is not supported on this platform")}
	}
	m, err := build()
	if err != nil {
		return &SharedImage{err: fmt.Errorf("wasm2go: building the shared memory image: %w", err)}
	}
	size := m.{{.MemSize}}.Load()
	dataEnd := m.{{.DataEnd}}
	if dataEnd == 0 || dataEnd > size {
		return &SharedImage{err: fmt.Errorf(
			"wasm2go: shared memory image: no data segments were installed (dataEnd=%d, size=%d)",
			dataEnd, size)}
	}

	// How much of the built instance the image keeps is the whole difference
	// between the two kinds.
	//
	// A snapshot keeps ALL of it, and its instances re-run nothing.
	//
	// A data-segment image keeps the segments and NOTHING above them, because
	// its instances DO re-run their initialization: that tail is BSS, and it
	// holds the start section's "already ran" flag and the C++ static state.
	// Inheriting the flag makes the start section trap on its second run;
	// inheriting statics that point into a heap this image does not carry
	// breaks the first thing that follows one of those pointers. The file
	// holds what is kept and nothing more: a mapping extends it to its
	// length (mmapImage), and everything past the end of the file reads as
	// zeros.
	keep := dataEnd
	if snapshot {
		keep = size
	}

	var globals []uint64
	if snapshot {
		// Globals live outside linear memory, so the image cannot carry them —
		// and the guest's TLS base and stack pointer are globals. Without them
		// the memory would describe an initialized instance that the Module
		// does not know it is.
		globals = {{.SaveGlobals}}(m)
	}

	f, err := newTempImageFile()
	if err != nil {
		return &SharedImage{err: fmt.Errorf("wasm2go: shared memory image: %w", err)}
	}
	if _, err := f.Write(m.{{.Memory}}[:keep]); err != nil {
		f.Close()
		return &SharedImage{err: fmt.Errorf("wasm2go: shared memory image: %w", err)}
	}
	return &SharedImage{f: f, size: size, dataEnd: dataEnd, globals: globals}
}

// AnonymousMemory returns memory for one instance that initializes itself
// on it: a private mapping, ceiling bytes long, that holds what from's
// memory holds below its data end and zeros above. from is an instance
// constructed and not run since: below its data end it holds the data
// segments, and whatever its start section (which a constructor runs) wrote
// there — the instance on the new memory runs its start section again, as
// one on a NewSharedImage image does. Build the instance with
// NewWithMemory (memory size InitialMemoryBytes) and initialize it as an
// instance on memory of its own: nothing is imaged or written to a file
// first, so the instance starts as fast as one that allocates its memory,
// and its memory keeps one address for its whole life, growing up to
// ceiling (native code that holds pointers into it needs that). Pages of
// from's memory that are all zeros are not copied: they stay untouched,
// costing nothing.
//
// The image returned stands for that memory in ForkMemory, as the image an
// instance's memory is a mapping of: the first fork of the instance writes
// its memory into an image and, where the platform can tell later what the
// instance changes, maps the memory onto that image (Rebased), so that the
// forks after it copy only what the instance changed. The image maps
// nothing itself. Pass the memory to UnmapMemory when the instance is done.
func AnonymousMemory(from *{{.Module}}, ceiling uint64) (*SharedImage, []byte, error) {
	if !mmapSupported {
		return nil, nil, fmt.Errorf("wasm2go: copy-on-write memory is not supported on this platform")
	}
	dataEnd := from.{{.DataEnd}}
	if dataEnd > uint64(len(from.{{.Memory}})) || dataEnd > ceiling {
		return nil, nil, fmt.Errorf("wasm2go: the data segments (%d bytes) do not fit in a memory of %d bytes", dataEnd, min(uint64(len(from.{{.Memory}})), ceiling))
	}
	length, err := mappingLength(ceiling)
	if err != nil {
		return nil, nil, err
	}
	mem, err := mmapAnonymous(length)
	if err != nil {
		return nil, nil, err
	}
	src := from.{{.Memory}}[:dataEnd]
	page := os.Getpagesize()
	zeros := make([]byte, page)
	for at := 0; at < len(src); at += page {
		end := min(at+page, len(src))
		if !bytes.Equal(src[at:end], zeros[:end-at]) {
			copy(mem[at:end], src[at:end])
		}
	}
	img := &SharedImage{dataEnd: dataEnd, anonymous: true,
		err: fmt.Errorf("wasm2go: memory of an instance's own is no image to map")}
	return img, mem, nil
}

// Err reports why the image is unavailable, or nil. A caller that gets an
// error must build each instance's memory itself.
func (s *SharedImage) Err() error { return s.err }

// Size is the guest's memory.size for the image, in bytes: what a Module built
// on it must be told its memory already holds.
func (s *SharedImage) Size() uint64 { return s.size }

// DataEnd is where the image's data segments stop and its BSS begins. For an
// image from NewSharedImage that is also where the image stops carrying data;
// a snapshot carries everything.
func (s *SharedImage) DataEnd() uint64 { return s.dataEnd }

// Globals are the snapshotted instance's wasm globals, to be handed to
// NewFromSnapshot along with the memory. Nil for a NewSharedImage image, which
// snapshots no instance and whose instances initialize their own globals.
func (s *SharedImage) Globals() []uint64 { return s.globals }

// Memory returns a private, copy-on-write mapping of the image, ceiling bytes
// long. The instance built on it must still run the start section and the
// module's initializers: memory.init will find the segments already in place
// and write nothing, while the initializers build this instance's own BSS.
//
// The ceiling is what the instance's memory can grow to, and costs address
// space only: no page of the mapping is charged to the process before it is
// written (see mapNoReserve). It is a uint64 because a memory64's memory
// may be longer than 4 GiB; a ceiling the process's address space cannot
// hold (above math.MaxInt) is refused with an error.
//
// The mapping is NOT Go heap — the GC will not reclaim it. Pass it to
// UnmapMemory when the instance is done.
func (s *SharedImage) Memory(ceiling uint64) ([]byte, error) {
	if s.err != nil {
		return nil, s.err
	}
	if ceiling < s.size {
		return nil, fmt.Errorf("wasm2go: memory ceiling %d is below the image (%d bytes)", ceiling, s.size)
	}
	length, err := mappingLength(ceiling)
	if err != nil {
		return nil, err
	}
	return mmapImage(s.f, length)
}

// Close releases the image itself. Mappings already handed out by Memory stay
// valid — each holds its own reference to the pages — but no further mapping
// can be made. An image that serves one instance (a snapshot taken to fork a
// running instance, say) is closed as soon as that instance is mapped, so the
// image's file does not outlive its only use. Idempotent.
func (s *SharedImage) Close() error {
	if s.f == nil {
		return nil
	}
	err := s.f.Close()
	s.f = nil
	if s.err == nil {
		s.err = fmt.Errorf("wasm2go: the shared memory image is closed")
	}
	return err
}

// MemoryFork is the duplicate of a running instance's linear memory that
// ForkMemory makes.
type MemoryFork struct {
	// Memory is a private mapping that holds what the instance's memory
	// held, for a second instance (NewFromSnapshot with Size and Globals).
	// Pass it to UnmapMemory when that instance is done.
	Memory []byte
	// Size is the instance's memory.size, in bytes.
	Size uint64
	// Globals are the instance's wasm globals.
	Globals []uint64
	// Image is the image Memory is a mapping of, for a later ForkMemory of
	// the instance built on Memory. nil where an image is of no use for
	// that (the platform cannot tell what an instance changed since).
	Image *SharedImage
	// Rebased reports that the forked instance's own memory was mapped
	// onto Image as well: Image, not the image it was given, is what a
	// later ForkMemory of it starts from.
	Rebased bool
}

// forkRebaseShare decides when a fork makes a new image: when the pages the
// instance changed since its image was made amount to more than one part in
// forkRebaseShare of its memory. Below that the changed pages are copied for
// each fork; above it one image is written, and what the following forks
// copy starts from nothing again.
const forkRebaseShare = 8

// ForkMemory duplicates the linear memory of a running instance, for an
// embedder that implements fork by giving a second instance the first one's
// state. The instance must not be executing: the caller is the goroutine
// that runs it (inside a host import, say).
//
// from is the image m's memory is a private mapping of (SharedImage.Memory,
// or the Memory of an earlier MemoryFork), the image AnonymousMemory
// returned with m's memory, or nil when the memory is none of these. Where the platform can tell which pages the instance has changed
// since (Linux), the duplicate is a second private mapping of the same image
// into which only those pages are copied, so a fork costs what the instance
// changed, not what it holds. When that is a large part of the memory, or
// cannot be told, the whole memory is written into a new image and the
// duplicate maps it; the instance's own memory is then mapped onto the new
// image too (Rebased), so that its next forks are cheap again; so is memory
// from AnonymousMemory at its first fork. Without from every fork writes the
// whole memory, and ceiling is the length of the duplicate's mapping; with
// from the duplicate is as long as m's.
//
// The image is shared by the mappings made of it. Close an image when no
// instance that may still be forked maps it.
func ForkMemory(m *{{.Module}}, from *SharedImage, ceiling uint64) (*MemoryFork, error) {
	if !mmapSupported {
		return nil, fmt.Errorf("wasm2go: copy-on-write memory is not supported on this platform")
	}
	// Writes from other goroutines (a signal's flag) go through
	// accessMemory: holding the lock keeps them out of the gap between the
	// copy and the remapping.
	m.{{.MemMu}}.Lock()
	defer m.{{.MemMu}}.Unlock()
	size := m.{{.MemSize}}.Load()
	mem := m.{{.Memory}}
	fork := &MemoryFork{Size: size, Globals: {{.SaveGlobals}}(m)}
	mapped := from != nil && from.f != nil
	anonymous := from != nil && from.anonymous
	if mapped || anonymous {
		// The memory is the first memory.size bytes of its mapping (all
		// of it, for a shared memory); the mapping is its capacity.
		mem = mem[:cap(mem)]
		ceiling = uint64(len(mem))
	}
	if ceiling < size {
		return nil, fmt.Errorf("wasm2go: memory ceiling %d is below the instance's memory (%d bytes)", ceiling, size)
	}
	length, err := mappingLength(ceiling)
	if err != nil {
		return nil, err
	}
	if mapped {
		if runs, changed, ok := privatePages(mem[:size]); ok && changed <= size/forkRebaseShare {
			child, err := mmapImage(from.f, length)
			if err != nil {
				return nil, err
			}
			for _, r := range runs {
				copy(child[r.from:r.to], mem[r.from:r.to])
			}
			fork.Memory, fork.Image = child, from
			return fork, nil
		}
	}
	f, err := newImageFile()
	if err != nil {
		return nil, fmt.Errorf("wasm2go: memory image: %w", err)
	}
	if _, err := f.Write(mem[:size]); err != nil {
		f.Close()
		return nil, fmt.Errorf("wasm2go: memory image: %w", err)
	}
	child, err := mmapImage(f, length)
	if err != nil {
		f.Close()
		return nil, err
	}
	fork.Memory = child
	if !privatePagesKnown {
		// Nothing can tell later what an instance changed since: the image
		// has served.
		if err := f.Close(); err != nil {
			unmapMemory(child)
			return nil, fmt.Errorf("wasm2go: memory image: %w", err)
		}
		return fork, nil
	}
	fork.Image = &SharedImage{f: f, size: size, dataEnd: m.{{.DataEnd}}}
	if (mapped || anonymous) && remapImage(f, mem) == nil {
		fork.Rebased = true
	}
	return fork, nil
}

// pageRun is a run of pages, as byte offsets into a mapping.
type pageRun struct{ from, to uint64 }

// mappingLength is a memory ceiling as the length of a mapping. A ceiling
// is a uint64, as a memory's size is, so that a memory64's ceiling above 4
// GiB can be named on any host; a host whose address space cannot hold it
// (a 32-bit one, for a ceiling above math.MaxInt) refuses it here, and the
// embedder falls back as it does for any mapping that fails.
func mappingLength(ceiling uint64) (int, error) {
	if ceiling > math.MaxInt {
		return 0, fmt.Errorf("wasm2go: a memory ceiling of %d bytes does not fit in this process's address space", ceiling)
	}
	return int(ceiling), nil
}

// newTempImageFile returns an empty temporary file that has no name any
// more — the image must never be reachable by name: the pages live as long
// as the descriptor or a mapping of it. The descriptor is one of the
// runtime's own, out of the way of a program's (raiseImageFile).
func newTempImageFile() (*os.File, error) {
	f, err := os.CreateTemp("", "wasm2go-image-*")
	if err != nil {
		return nil, err
	}
	_ = os.Remove(f.Name())
	return raiseImageFile(f), nil
}

// UnmapMemory releases a mapping returned by SharedImage.Memory. Calling it on
// a slice that did not come from there is a no-op on platforms without mmap
// and undefined otherwise, so do not.
func UnmapMemory(mem []byte) { unmapMemory(mem) }
